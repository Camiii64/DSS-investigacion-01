DROP DATABASE IF EXISTS hospital_DB;
CREATE DATABASE hospital_DB;
USE hospital_DB;

-- ==========================
-- TABLA USUARIOS (LOGIN)
-- ==========================
CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    rol ENUM('ADMIN','DOCTOR','PACIENTE') NOT NULL,
    estado BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ==========================
-- TABLA DOCTORES
-- ==========================
CREATE TABLE doctores (
    id INT PRIMARY KEY,
    especialidad VARCHAR(100) NOT NULL,
    licencia_medica VARCHAR(50) NOT NULL UNIQUE,
    FOREIGN KEY (id) REFERENCES usuarios(id)
        ON DELETE CASCADE
);

-- ==========================
-- TABLA PACIENTES
-- ==========================
CREATE TABLE pacientes (
    id INT PRIMARY KEY,
    fecha_nacimiento DATE NOT NULL,
    telefono VARCHAR(15),
    tipo_sangre ENUM('A+','A-','B+','B-','AB+','AB-','O+','O-'),
    FOREIGN KEY (id) REFERENCES usuarios(id)
        ON DELETE CASCADE
);

-- ==========================
-- TABLA CITAS
-- ==========================
CREATE TABLE citas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    paciente_id INT NOT NULL,
    doctor_id INT NOT NULL,
    fecha_solicitada DATETIME NOT NULL,
    motivo TEXT NOT NULL,
    estado ENUM('PENDIENTE','ACEPTADA','RECHAZADA','CANCELADA')
        DEFAULT 'PENDIENTE',
    respuesta_doctor TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (paciente_id) REFERENCES pacientes(id)
        ON DELETE CASCADE,
    FOREIGN KEY (doctor_id) REFERENCES doctores(id)
        ON DELETE CASCADE
);
